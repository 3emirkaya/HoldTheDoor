\begin{itemize}[label=\textbullet]
  \item Basic

    \item Puzzle
    \item Role-play
    \item Simulator
    \item Strategy
  \end{itemize}
  \item More Specific
  \begin{itemize}[label=$\circ$]

      \item MMORPG (role-playing game)
      \item MMORTS (real-time strategy)

    \item Puzzle
    \item Racing
    \item RPG (role-playing game)

    \item Shoot-em-up (scrolling shooter)

    \item Trading
